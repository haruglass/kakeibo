// ========================================== // FORM MANAGEMENT // ========================================== /** * Updates the visibility of standard/transfer account fields * and populates the category dropdown based on the transaction type. */ function updateForm() { const type = document.getElementById('type').value; const isTransfer = type === '振替'; // Toggle field visibilities using a cleaner classList API document.getElementById('transfer-accounts').classList.toggle('display-none', !isTransfer); document.getElementById('standard-account').classList.toggle('display-none', isTransfer); document.getElementById('category-field').classList.toggle('display-none', isTransfer); // Repopulate categories dynamically const catSelect = document.getElementById('category'); catSelect.innerHTML = ''; if (!isTransfer && categories[type]) { categories[type].forEach(category => { catSelect.add(new Option(category, category)); }); } } // ========================================== // DATA LOADING & DOM RENDERING // ========================================== /** * Fetches transactional data from GAS, parses it safely, * and coordinates the updates for Balances, Statistics, and Filters. */ async function loadData() { const loadingEl = document.getElementById('loading'); loadingEl.style.display = 'flex'; try { const response = await fetch(GAS_URL); if (!response.ok) throw new Error(`Network response error: ${response.status}`); const data = await response.json(); // Cache global states currentStatsData = data.statsBody || []; currentHistoryData = data.history || []; // Core data rendering orchestrations updateMonthFilterOptions(currentStatsData); applyFilter(); renderBalanceSheet(data.balances || []); renderStatisticsDashboard(currentStatsData); } catch (error) { console.error("Failed to load spreadsheet data:", error); alert("データの読み込みに失敗しました。"); } finally { loadingEl.style.display = 'none'; } } /** * Generates the unified balance tracking DOM nodes */ function renderBalanceSheet(balances) { const balanceList = document.getElementById('balance-list'); balanceList.innerHTML = balances.map(([accountName, balanceValue]) => `
${accountName} ¥${Number(balanceValue || 0).toLocaleString()}
`).join(''); } /** * Generates the aggregated monthly overview metrics card matrix */ function renderStatisticsDashboard(statsBody) { const statsList = document.getElementById('stats-list'); statsList.innerHTML = statsBody.map((row) => { const ym = extractYearMonthStr(row[0]); const displayLabel = ym ? `${ym.year.substring(2)}年${Number(ym.month)}月` : row[0]; // Pre-calculate indexes mapping directly to schema const totalExpenses = row[1]; const totalIncomes = row[11]; const netBalance = row[17]; return `
${displayLabel}
¥${totalExpenses.toLocaleString()}
収入 ¥${totalIncomes.toLocaleString()} 収支 ¥${netBalance.toLocaleString()}
${expenseLabels.map((label, idx) => `
${label}¥${row[idx + 2].toLocaleString()}
`).join('')} ${incomeLabels.map((label, idx) => `
${label}¥${row[idx + 12].toLocaleString()}
`).join('')}
`; }).join(''); } /** * Populates unique selectable target milestones inside the Month Filter drop-down */ function updateMonthFilterOptions(statsBody) { const monthSelect = document.getElementById('filter-month'); const currentSelectedValue = monthSelect.value; monthSelect.innerHTML = ''; statsBody.forEach(row => { const ym = extractYearMonthStr(row[0]); if (ym) { const optionExists = Array.from(monthSelect.options).some(opt => opt.value === ym.key); if (!optionExists) { const displayLabel = `${ym.year.substring(2)}年${Number(ym.month)}月`; monthSelect.add(new Option(displayLabel, ym.key)); } } }); // Re-assign previous selection state seamlessly if it matches existing indexes monthSelect.value = currentSelectedValue || "all"; }
